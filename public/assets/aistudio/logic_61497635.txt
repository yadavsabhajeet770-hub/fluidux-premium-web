// ── Helper: Convert number to B/S ──
function numToType(n) {
    return n >= 5 ? 'B' : 'S';

// ── Helper: Look up pattern in DB ──
function lookupPattern(pattern) {
    return CPU_PATTERN_DB[pattern] || null;
}

// ── Helper: Live adaptive detection (fallback) ──
function liveAdaptiveDetect(types) {
    if (types.length < 4) return null;
    const len = Math.min(types.length, 10);
    const slice = types.slice(0, len);
    const seq = slice.join('');

    // Try to find longest matching pattern in DB
    for (let l = Math.min(seq.length, 9); l >= 3; l--) {
        const sub = seq.substring(0, l);
        const hit = lookupPattern(sub);
        if (hit) {
            return {
                method: 'AI_LIVE',
                size: hit.next,
                conf: hit.conf
            };
        }
    }

    // Fallback: statistical
    const bCount = slice.filter(t => t === 'B').length;
    const total = slice.length;
    const bRatio = bCount / total;
    if (bRatio >= 0.7) {
        return { method: 'MATH', size: 'SMALL', conf: 65 + Math.floor(bRatio * 20) };
    } else if (bRatio <= 0.3) {
        return { method: 'MATH', size: 'BIG', conf: 65 + Math.floor((1 - bRatio) * 20) };
    }
    return null;
}

                }
            }

            if (!found) {
                patKey = 'CHAOS';
                conf = 55;
                pred = Math.random() < 0.5 ? 'BIG' : 'SMALL';
            }

    const info = BUNNY_PATTERN_NAMES[patKey] || ['Unknown', '🔍 Analyzing...'];
    const patName = info[0];
    let msg = info[1];

    const confCol = conf >= 85 ? '#22c55e' : conf >= 70 ? '#facc15' : '#f97316';

    const patEl = document.getElementById('bunnyPatName');

        callEl.style.color = callCol; }

    // ── Store for external use ──
    currentPrediction = { size: pred, confidence: conf, pattern: patKey };

    // ── Return result ──
    return {
        pattern: patKey,
        patternName: patName,
        prediction: pred,
        confidence: conf,
        message: msg
    };
